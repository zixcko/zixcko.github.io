📥 Cara 1: Unduh Tabel untuk Excel / Google Spreadsheet

Paling gampang! Tekan tombol di bawah untuk mengunduh semua catatan menjadi file tabel (`.CSV`) yang bisa langsung dibuka di Microsoft Excel atau Google Spreadsheet.

☁️ Cara 2: Sambungkan Otomatis ke Google Spreadsheet (Google Script)

Jika website ini dipasang di Google Apps Script atau dihubungkan ke Google Spreadsheet, tempelkan alamat Web App Google Script Anda di bawah ini:

🔧 Lihat Kode Google Apps Script (`Code.gs`) & Rencana Upgrade Database SQL (Untuk Teknisi/Admin)
Code.gs (Pasang di Google Spreadsheet → Ekstensi → Apps Script) 
const SHEET_CATATAN = 'CatatanKas';

function doGet(e) {
  if (e && e.parameter && e.parameter.action === 'ambil') {
    return ContentService.createTextOutput(JSON.stringify(ambilDataSheet()))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return HtmlService.createHtmlOutputFromFile('buku_kas_umkm')
    .setTitle('BukuKas Pintar UMKM')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function doPost(e) {
  const body = JSON.parse(e.postData.contents);
  if (body.action === 'simpanSemua') {
    simpanKeSheet(body.records);
    return ContentService.createTextOutput(JSON.stringify({ status: 'sukses' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_CATATAN);
  if (!sh) {
    sh = ss.insertSheet(SHEET_CATATAN);
    sh.appendRow(['id', 'tanggal', 'jenis', 'kantong', 'kategori', 'judul', 'nominal', 'caraBayar']);
  }
  return sh;
}

function ambilDataSheet() {
  const sh = getOrCreateSheet();
  const rows = sh.getDataRange().getValues();
  const headers = rows.shift();
  return rows.map(r => {
    let item = {};
    headers.forEach((h, i) => item[h] = r[i]);
    item.nominal = Number(item.nominal) || 0;
    return item;
  });
}

function simpanKeSheet(records) {
  const sh = getOrCreateSheet();
  sh.clearContents();
  const headers = ['id', 'tanggal', 'jenis', 'kantong', 'kategori', 'judul', 'nominal', 'caraBayar'];
  sh.appendRow(headers);
  if (records && records.length) {
    const rows = records.map(r => headers.map(h => r[h] || ''));
    sh.getRange(2, 1, rows.length, headers.length).setValues(rows);
  }
  return true;
}
